# ============================================================
# STS ML SCORE — lower chart study (daily)
# The same learned-weight score as the column/scan, computed on
# EVERY bar — so you see the score's history alongside price.
#
# INSTALL: Charts -> Studies -> Edit Studies -> Create ->
# paste -> OK. Shows as a lower pane. Use a DAILY chart with
# 2 years+ of data (the 52-week lookback needs runway; the
# left ~1 year of the plot will be blank — normal).
#
# READING THE LINE (with the levels):
#   above 2.0 (green)  : D10 zone — the validated trade bucket
#   1.0-2.0  (yellow)  : top-quintile fringe / watchlist
#   below 1.0 (gray)   : not a candidate
# The SLOPE is the insight the column can't show:
#   RISING score  = markdown deepening while long-term trend
#                   holds -> setup building
#   FALLING score = stock recovering toward its high (or trend
#                   breaking) -> setup spending itself
# A fresh CROSS above 2.0 = the day the stock would first have
# appeared on your scan.
# ============================================================

declare lower;

def atr14   = MovingAverage(AverageType.WILDERS, TrueRange(high, close, low), 14);
def high52  = close / Highest(high, 252) - 1;
def dist200 = (close - Average(close, 200)) / atr14;
def mom6    = close[10] / close[126] - 1;
def rsi20   = RSI(length = 20, price = close);
def qqe     = (ExpAverage(rsi20, 5) - 50) / 50;
def hma     = HullMovingAvg(close, 20);
def hullSl  = (hma - hma[1]) / atr14;
def mom40   = close / close[40] - 1;

script zclip { input v = 0; plot z = Max(-3, Min(3, v)); }
def zHigh52 = zclip((high52  - (-0.149772)) / 0.156737);
def zDist   = zclip((dist200 -   1.681650 ) / 6.240695);
def zMom6   = zclip((mom6    -   0.086250 ) / 0.361377);
def zQqe    = zclip((qqe     -   0.044473 ) / 0.194268);
def zHull   = zclip((hullSl  -   0.027496 ) / 0.303103);
def zMom40  = zclip((mom40   -   0.020000 ) / 0.155298);

plot Score = -2.20 * zHigh52 + 1.35 * zDist + 1.08 * zMom6
           + -0.99 * zQqe + 0.63 * zMom40 + 0.60 * zHull;
Score.SetLineWeight(2);
Score.AssignValueColor(
    if Score >= 2.0 then Color.GREEN
    else if Score >= 1.0 then Color.YELLOW
    else Color.GRAY);

plot D10line  = 2.0;
D10line.SetDefaultColor(Color.DARK_GREEN);
D10line.SetStyle(Curve.SHORT_DASH);
plot Qline = 1.0;
Qline.SetDefaultColor(Color.DARK_GRAY);
Qline.SetStyle(Curve.SHORT_DASH);
plot Zero = 0;
Zero.SetDefaultColor(Color.DARK_GRAY);
