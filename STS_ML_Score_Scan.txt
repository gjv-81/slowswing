# ============================================================
# STS ML SCORE — scan filter (daily)
# Same learned-weight score as the watchlist column, as a scan
# condition. Returns stocks above the threshold; save the scan
# results as a dynamic watchlist ("STS ML D10").
#
# INSTALL (GREEN-RING setup — scan the whole market, validated
# population only):
#   Scan in: All Stocks
#   Stock filter 1: Market Cap  min 10,000  ($10B — edge validated
#                   only in MEGA/LARGE per the cap-tier audit)
#   Stock filter 2: Last  min 15
#   Study filter:   this script (Aggregation DAY) — it enforces
#                   score >= threshold AND $70M/day dollar volume
# Listing age: names under ~1 year self-exclude (52-wk features
# return N/A). 1-3 year names pass — treat as amber (SSI2 heavier).
#
# THRESHOLD GUIDE (from the 10-year research panel, pooled):
#   score >= 2.0  ~ top decile (D10)     - the tested trade bucket
#   score >= 1.0  ~ top quintile (D9-10)
# The true decile is a daily RANK, which a scan cannot compute;
# a fixed threshold drifts with market conditions. The sortable
# column is the faithful method - use this scan as a first pass,
# then rank the results by the STS_ML column.
# ============================================================

input threshold = 2.0;
input minDollarVolume = 70000000;   # $70M/day = 5th pctile of training universe

def atr14   = MovingAverage(AverageType.WILDERS, TrueRange(high, close, low), 14);

def high52  = close / Highest(high, 252) - 1;
def dist200 = (close - Average(close, 200)) / atr14;
def mom6    = close[10] / close[126] - 1;
def rsi20   = RSI(length = 20, price = close);
def qqe     = (ExpAverage(rsi20, 5) - 50) / 50;
def hma     = HullMovingAvg(close, 20);
def hullSl  = (hma - hma[1]) / atr14;
def mom40   = close / close[40] - 1;

script zclip { input v = 0; plot z = Max(-3, Min(3, v)); }
def zHigh52 = zclip((high52  - (-0.149772)) / 0.156737);
def zDist   = zclip((dist200 -   1.681650 ) / 6.240695);
def zMom6   = zclip((mom6    -   0.086250 ) / 0.361377);
def zQqe    = zclip((qqe     -   0.044473 ) / 0.194268);
def zHull   = zclip((hullSl  -   0.027496 ) / 0.303103);
def zMom40  = zclip((mom40   -   0.020000 ) / 0.155298);

def score = -2.20 * zHigh52
          +  1.35 * zDist
          +  1.08 * zMom6
          + -0.99 * zQqe
          +  0.63 * zMom40
          +  0.60 * zHull;

def liquid = Average(volume, 20) * close >= minDollarVolume;

plot scan = score >= threshold and liquid;
